func readNativeCombatFixture(t *testing.T, kind string) []byte {
	t.Helper()
	path := filepath.Join("..", "..", "..", "data", "test-fixtures", "combat-native-"+kind+".json")
	data, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	return data
}

// Fixtures execute the original DLL's command writers, framing, checksum and
// receive preprocessing in an isolated i386 process. Regenerate with
// scripts/native-combat-fixture.py; they are independent of the Go encoder.
func TestCombatDataMatchesNativeFixtures(t *testing.T) {
	for _, kind := range []string{"title", "mech", "ready", "commando", "position", "client-position", "fire", "server-fire", "targeted-fire", "server-targeted-fire", "damage", "local-damage"} {
		t.Run(kind, func(t *testing.T) {
			data := readNativeCombatFixture(t, kind)
			var fixture struct {
				CommandHex          string `json:"command_hex"`
				FrameHex            string `json:"server_frame_hex"`
				ClientChecksum      uint32 `json:"client_checksum"`
				ServerChecksum      uint32 `json:"server_checksum"`
				NativeDecodeMatches bool   `json:"native_decode_matches"`
			}
			if err := json.Unmarshal(data, &fixture); err != nil {
				t.Fatal(err)
			}
			command, err := hex.DecodeString(fixture.CommandHex)
			if err != nil {
				t.Fatal(err)
			}
			frame, err := hex.DecodeString(fixture.FrameHex)
			if err != nil {
				t.Fatal(err)
			}
			if !fixture.NativeDecodeMatches {
				t.Fatal("fixture did not round-trip through the native preprocessor")
			}
			pdu := CombatData(command)
			want := append(frame, 0)
			if pdu.ID != CombatDataID || binary.BigEndian.Uint32(pdu.Body) != uint32(len(want)) || !bytes.Equal(pdu.Body[4:], want) {
				t.Fatalf("native combat frame mismatch: got %x, want counted %x", pdu.Body, want)
			}
			if combatChecksum(command, true) != fixture.ServerChecksum || combatChecksum(command, false) != fixture.ClientChecksum {
				t.Fatal("checksum differs from the native instruction result")
			}
			if kind == "ready" && !bytes.Equal(CombatReadyCommand(), command) {
				t.Fatalf("ready command differs from native writer: %x", command)
			}
		})
	}
}

func TestNativeClientPositionMapsToNativeServerPosition(t *testing.T) {
	fixtureCommand := func(kind string) []byte {
		t.Helper()
		data := readNativeCombatFixture(t, kind)
		var fixture struct {
			CommandHex string `json:"command_hex"`
		}
		if err := json.Unmarshal(data, &fixture); err != nil {
			t.Fatal(err)
		}
		command, err := hex.DecodeString(fixture.CommandHex)
		if err != nil {
			t.Fatal(err)
		}
		return command
	}
	client := fixtureCommand("client-position")
	want := fixtureCommand("position")
	got, err := CombatPeerPosition(1, client)
	if err != nil {
		t.Fatal(err)
	}
	if !bytes.Equal(got, want) {
		t.Fatalf("native peer position differs: got %x, want %x", got, want)
	}
	client[1] = 0
	if _, err := CombatPeerPosition(1, client); err == nil {
		t.Fatal("accepted malformed native position digit")
	}
}

func TestCombatCommandWritersMatchNativeFixtures(t *testing.T) {
	for _, kind := range []string{"title", "commando"} {
		t.Run(kind, func(t *testing.T) {
			data := readNativeCombatFixture(t, kind)
			var fixture struct {
				CommandHex string `json:"command_hex"`
			}
			if err := json.Unmarshal(data, &fixture); err != nil {
				t.Fatal(err)
			}
			want, err := hex.DecodeString(fixture.CommandHex)
			if err != nil {
				t.Fatal(err)
			}
			writer := combatWriter{}
			if kind == "title" {
				writer.title("Fixture", 1, 1, [4]uint32{1729, 11, 22, 33}, 44, [4]byte{2, 3, 4, 55})
			} else {
				writer.commando(CombatPilot{ID: 0, Team: 1, CallSign: "FixturePilot", X: 10000, Y: 20000})

				t.Fatal(writer.err)
			}

				t.Fatalf("command differs from native writer:\ngot  %x\nwant %x", writer.data, want)
			}
		})
	}
}

func TestCombatBootstrapDeclaresExactMechCountAndDefersReady(t *testing.T) {
	setup, err := CombatBootstrap(0, []CombatPilot{
		{ID: 0, Team: 0, CallSign: "Subvoltic", X: 10000, Y: 100000},
		{ID: 1, Team: 1, CallSign: "BetaPilot", X: 12000, Y: 100000},
	})
	if err != nil {
		t.Fatal(err)
	}
	// After u32 length, ESC and prefix, Title serializes name, local ID,
	// climate, four three-digit values, then two-digit heat. Its next byte
	// is the exact Mech count checked at native AddNewMech:0x690576d7.
	command := setup.Body[6 : len(setup.Body)-5]
	countOffset := 1 + 1 + len("Solaris") + 2 + 4*3 + 2
	if command[countOffset] != 2+'!' {
		t.Fatalf("declared native Mech count %d, want 2", command[countOffset]-'!')
	}
	if command[len(command)-1] == '_' {
		t.Fatal("Ready was sent before native PreCombatInit can initialize HUD nodes")
	}
}

	// First real position packet after the two-Mech setup, captured from
	// native-bootstrap-stage3 at 13:47:33. No terminal NUL is counted.
	body, err := hex.DecodeString("0000002b1b21274b4d2e5a4b593d7322302b4b4b4b4b4b4b4b4d4e4b4b4b4b4b4b4b4b4b4b4b4b2139574b3859341b")

		t.Fatal(err)
	}
	command, err := ParseClientCombatData(body)
	if err != nil {
		t.Fatal(err)
	}
	if len(command) != 37 || command[0] != 0x27 {

		t.Fatal("accepted corrupted native combat data")
	}
}

func TestNativeTargetedFireReferences(t *testing.T) {
	commands := make(map[string][]byte)
	for _, kind := range []string{"targeted-fire", "server-targeted-fire"} {
		var fixture struct {
			CommandHex string `json:"command_hex"`
		}
		if err := json.Unmarshal(readNativeCombatFixture(t, kind), &fixture); err != nil {
			t.Fatal(err)
		}
		command, err := hex.DecodeString(fixture.CommandHex)
		if err != nil {
			t.Fatal(err)
		}
		commands[kind] = command
	}
	client := commands["targeted-fire"]
	selection, err := ParseClientCombatFire(client)
	if err != nil || selection.WeaponSlot != 2 || selection.TargetEntity != 1 {
		t.Fatalf("native targeted selection = %+v, %v", selection, err)
	}
	peer, err := CombatPeerFire(1, client)
	if err != nil || !bytes.Equal(peer, commands["server-targeted-fire"]) {
		t.Fatalf("targeted fire differs from native server writer: %x, %v", peer, err)
	}
	client[2] = '!'
	selection, err = ParseClientCombatFire(client)
	if err != nil || selection.TargetEntity != -1 {
		t.Fatal("zero target must decode as untargeted")
	}
	client[35] = 0
	if _, err := ParseClientCombatFire(client); err == nil {
		t.Fatal("accepted malformed fire tail")
	}
	if _, err := ParseClientCombatFire(client[:2]); err == nil {
		t.Fatal("accepted truncated fire command")
	}
}

func TestLiveFireReportIncludesNativeHitReturnPoint(t *testing.T) {
	var fixture struct {
		Shots []struct {
			ShooterEntity int    `json:"shooter_entity"`
			ClientBodyHex string `json:"client_body_hex"`
		} `json:"shots"`
	}
	data, err := os.ReadFile(filepath.Join("..", "..", "..", "data", "test-fixtures", "combat-live-armor-probe.json"))
	if err != nil {
		t.Fatal(err)
	}
	if err := json.Unmarshal(data, &fixture); err != nil {
		t.Fatal(err)
	}
	want := [][3]int32{{-9936, 13891, 0}, {9930, 13891, 0}}
	wantPosition := []CombatPosition{{X: 0, Y: 10000}, {X: 0, Y: -10000}}
	if len(fixture.Shots) != len(want) {
		t.Fatalf("live shot count = %d", len(fixture.Shots))
	}
	for i, shot := range fixture.Shots {
		body, err := hex.DecodeString(shot.ClientBodyHex)
		if err != nil {
			t.Fatal(err)
		}
		commands, err := ParseClientCombatData(body)
		if err != nil {
			t.Fatalf("shooter %d combat frame: %v", shot.ShooterEntity, err)
		}
		batch, err := ParseClientCombatCommands(commands)
		if err != nil || len(batch.Fires) != 1 {
			t.Fatalf("shooter %d batch = %+v, %v", shot.ShooterEntity, batch, err)
		}
		position, err := ParseClientCombatPositionCommand(batch.Position)
		if err != nil || position != wantPosition[i] {
			t.Fatalf("shooter %d position = %+v, %v; want %+v", shot.ShooterEntity, position, err, wantPosition[i])
		}
		selection, err := ParseClientCombatFire(batch.Fires[0])
		if err != nil || !selection.HasHitPoint || selection.HitPoint != want[i] {
			t.Fatalf("shooter %d hit point = %+v, %v; want %v", shot.ShooterEntity, selection, err, want[i])
		}
	}
}

func TestNativeStandaloneFireAndCompleteBatches(t *testing.T) {
	data, err := os.ReadFile(filepath.Join("..", "..", "..", "data", "test-fixtures", "combat-live-standalone-fire.json"))
	if err != nil {
		t.Fatal(err)
	}
	var fixture struct {
		ClientBodyHex string `json:"client_body_hex"`
	}
	if err := json.Unmarshal(data, &fixture); err != nil {
		t.Fatal(err)
	}
	body, err := hex.DecodeString(fixture.ClientBodyHex)
	if err != nil {
		t.Fatal(err)
	}
	fire, err := ParseClientCombatData(body)
	if err != nil {
		t.Fatal(err)
	}
	batch, err := ParseClientCombatCommands(fire)
	if err != nil || batch.Position != nil || len(batch.Fires) != 1 {
		t.Fatalf("standalone fire = %+v, %v", batch, err)
	}
	if _, prepared := ClientCombatPosition(fire); prepared {
		t.Fatal("standalone fire released preparation barrier")
	}
	var positionFixture struct {
		CommandHex string `json:"command_hex"`
	}
	if err := json.Unmarshal(readNativeCombatFixture(t, "client-position"), &positionFixture); err != nil {
		t.Fatal(err)
	}
	position, err := hex.DecodeString(positionFixture.CommandHex)
	if err != nil {
		t.Fatal(err)
	}
	combined := append(append(append([]byte(nil), position...), fire...), fire...)
	batch, err = ParseClientCombatCommands(combined)
	if err != nil || len(batch.Position) != 37 || len(batch.Fires) != 2 {
		t.Fatalf("position and two fire events = %+v, %v", batch, err)
	}
	for _, bad := range [][]byte{nil, fire[:35], combined[:len(combined)-1], append(append([]byte(nil), position...), '_'), append(append([]byte(nil), fire...), position...)} {
		batch, err := ParseClientCombatCommands(bad)
		if err == nil || batch.Position != nil || len(batch.Fires) != 0 {
			t.Fatal("unsupported/truncated batch produced partial commands")
		}
	}
}

func TestLiveTargetedCombatFireBatch(t *testing.T) {
	data, err := os.ReadFile(filepath.Join("..", "..", "..", "data", "test-fixtures", "combat-live-targeted-fire.json"))
	if err != nil {
		t.Fatal(err)
	}
	var fixture struct {
		Shots []struct {
			Body   string `json:"client_body_hex"`
			Weapon byte   `json:"weapon_slot"`
			Target int    `json:"target_entity"`
		} `json:"shots"`
	}
	if err := json.Unmarshal(data, &fixture); err != nil {
		t.Fatal(err)
	}
	if len(fixture.Shots) != 2 {
		t.Fatal("expected the two opposing native laser reports")
	}
	for i, shot := range fixture.Shots {
		body, err := hex.DecodeString(shot.Body)
		if err != nil {
			t.Fatal(err)
		}
		commands, err := ParseClientCombatData(body)
		if err != nil {
			t.Fatal(err)
		}
		batch, err := ParseClientCombatCommands(commands)
		if err != nil || len(batch.Position) != 37 || len(batch.Fires) != 1 {
			t.Fatalf("shot %d: incomplete native position/fire batch: %v", i, err)
		}
		selection, err := ParseClientCombatFire(batch.Fires[0])
		if err != nil || selection.WeaponSlot != shot.Weapon || selection.TargetEntity != shot.Target || selection.HitPart != 4 || shot.Target != 1-i {
			t.Fatalf("shot %d: incorrect opposing target: %+v, %v", i, selection, err)
		}
		peer, err := CombatPeerFire(byte(i), batch.Fires[0])
		if err != nil || peer[0] != 'e' || peer[1] != byte(i)+'!' || !bytes.Equal(peer[2:], batch.Fires[0][1:35]) {
			t.Fatalf("shot %d changed when converted for its peer", i)
		}
	}
}

func TestNativeDamageWriterLayouts(t *testing.T) {
	for _, tc := range []struct{ kind, command string }{
		{"damage", "c\"#&!$"},
		{"local-damage", "d#&!$"},
	} {
		var fixture struct {
			CommandHex string `json:"command_hex"`
		}
		if err := json.Unmarshal(readNativeCombatFixture(t, tc.kind), &fixture); err != nil {
			t.Fatal(err)
		}
		command, err := hex.DecodeString(fixture.CommandHex)
		if err != nil || string(command) != tc.command {
			t.Fatalf("%s native writer changed: %x (%v)", tc.kind, command, err)
		}
	}
}

func TestNativeFireReportsCollisionPart(t *testing.T) {
	var fixture struct {
		CommandHex string `json:"command_hex"`
	}
	if err := json.Unmarshal(readNativeCombatFixture(t, "targeted-fire"), &fixture); err != nil {
		t.Fatal(err)
	}
	command, err := hex.DecodeString(fixture.CommandHex)
	if err != nil {
		t.Fatal(err)
	}
	report, err := ParseClientCombatFire(command)
	if err != nil || report.HitPart != 1 {
		t.Fatalf("native part: %+v, %v", report, err)
	}
	command[3], command[4], command[5] = '!', '!', '!'
	report, err = ParseClientCombatFire(command)
	if err != nil || report.HitPart != -1 {
		t.Fatal("zero part must decode as none")
	}
	command[3], command[4], command[5] = '"', '#', '$'
	report, err = ParseClientCombatFire(command)
	if err != nil || report.HitPart != 85*85+2*85+3-1 {
		t.Fatal("part must use all three unsigned base85 digits")
	}
}
