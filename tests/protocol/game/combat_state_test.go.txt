package protocol

import (
	"encoding/json"
	"math"
	"os"
	"path/filepath"
	"testing"
)

func TestCombatPositionWorldAxesMatchOriginalVisitor(t *testing.T) {
	data, err := os.ReadFile(filepath.Join("..", "..", "..", "data", "test-fixtures", "combat-native-state.json"))
	if err != nil {
		t.Fatal(err)
	}
	var f struct {
		Pose struct {
			X      int32      `json:"wire_x"`
			Y      int32      `json:"wire_y"`
			Height int32      `json:"wire_height"`
			World  [3]float64 `json:"world_xyz"`
		} `json:"peer_position"`
	}
	if err = json.Unmarshal(data, &f); err != nil {
		t.Fatal(err)
	}
	w := combatWriter{data: []byte{'\x27'}}
	w.coordinate(f.Pose.X)
	w.coordinate(f.Pose.Y)
	w.number(uint32(f.Pose.Height), 3)
	for range 25 {
		w.data = append(w.data, '!')
	}
	p, err := ParseClientCombatPositionCommand(w.data)
	if err != nil {
		t.Fatal(err)
	}
	point := p.WorldPoint()
	for i, v := range point {
		if math.Abs(float64(v)/100-f.Pose.World[i]) > 0.0001 {
			t.Fatalf("axis%d: %v vs native %v", i, point, f.Pose.World)
		}
	}
}
