func TestNativeCombatWaitsForBothPreparedClients(t *testing.T) {
	hub := newSessionHub()
	hub.experimentalCombatInit = true
	hub.combatPreparing = map[int]map[string]bool{1: {"Alpha": false, "Beta": false}}
	var sent []protocol.PDU
	collect := func(p protocol.PDU) error { sent = append(sent, p); return nil }
	alpha, beta, other := &liveSession{sendGamePDU: collect}, &liveSession{sendGamePDU: collect}, &liveSession{sendGamePDU: collect}
	hub.sessions[alpha] = planetPresence{callSign: "Alpha", location: state.Location{RoomID: 1}}
	hub.sessions[beta] = planetPresence{callSign: "Beta", location: state.Location{RoomID: 1}}
	hub.sessions[other] = planetPresence{callSign: "Other", location: state.Location{RoomID: 2}}
	frame, err := hex.DecodeString("0000002b1b21274b4d2e5a4b593d7322302b4b4b4b4b4b4b4b4d4e4b4b4b4b4b4b4b4b4b4b4b4b2139574b3859341b")
	if err != nil {
		t.Fatal(err)
	}
	hub.noteCombatPrepared(alpha, frame)
	hub.noteCombatPrepared(alpha, frame)
	hub.noteCombatPrepared(other, frame)
	if len(sent) != 0 {
		t.Fatal("combat released before both mission clients prepared")
	}
	bad := append([]byte(nil), frame...)
	bad[10] ^= 1
	hub.noteCombatPrepared(beta, bad)
	if len(sent) != 0 {
		t.Fatal("corrupt input released combat")
	}
	hub.noteCombatPrepared(beta, frame)
	if len(sent) != 2 {
		t.Fatalf("ready recipients %d, want only the two mission clients", len(sent))
	}
	want := protocol.CombatData(protocol.CombatReadyCommand())
	for _, p := range sent {
		if p.ID != want.ID || !bytes.Equal(p.Body, want.Body) {
			t.Fatal("unexpected native ready data")
		}
	}
	hub.noteCombatPrepared(beta, frame)
	if len(sent) != 2 {
		t.Fatal("duplicate native ready release")
	}
}

func TestNativeCombatPositionRelaysOnlyToActiveRoomPeer(t *testing.T) {
	hub := newSessionHub()
	hub.experimentalCombatInit = true
	hub.combatEntities = map[int]map[string]byte{1: {"Alpha": 0, "Beta": 1}, 2: {"Other": 0}}
	var alphaSent, betaSent, otherSent []protocol.PDU
	alpha := &liveSession{sendGamePDU: func(p protocol.PDU) error { alphaSent = append(alphaSent, p); return nil }}
	beta := &liveSession{sendGamePDU: func(p protocol.PDU) error { betaSent = append(betaSent, p); return nil }}
	other := &liveSession{sendGamePDU: func(p protocol.PDU) error { otherSent = append(otherSent, p); return nil }}
	hub.sessions[alpha] = planetPresence{callSign: "Alpha", location: state.Location{RoomID: 1}}
	hub.sessions[beta] = planetPresence{callSign: "Beta", location: state.Location{RoomID: 1}}
	hub.sessions[other] = planetPresence{callSign: "Other", location: state.Location{RoomID: 2}}
	frame, err := hex.DecodeString("0000002b1b21274b4d2e5a4b593d7322302b4b4b4b4b4b4b4b4d4e4b4b4b4b4b4b4b4b4b4b4b4b2139574b3859341b")
	if err != nil {
		t.Fatal(err)
	}
	hub.relayCombatCommands(alpha, frame)
	if len(betaSent) != 0 {
		t.Fatal("relayed position before both clients prepared")
	}
	hub.combatActive = map[int]bool{1: true}
	precheck, precheckErr := protocol.ParseClientCombatData(frame)
	if precheckErr != nil {
		t.Fatal(precheckErr)
	}
	if _, precheckErr = protocol.CombatPeerPosition(0, precheck); precheckErr != nil {
		t.Fatalf("captured native position cannot relay: %v, %x", precheckErr, precheck)
	}
	hub.relayCombatCommands(alpha, frame)
	if len(alphaSent) != 0 || len(otherSent) != 0 || len(betaSent) != 1 {
		t.Fatalf("wrong room-scoped recipients: alpha=%d beta=%d other=%d", len(alphaSent), len(betaSent), len(otherSent))
	}
	clientCommand, err := protocol.ParseClientCombatData(frame)
	if err != nil {
		t.Fatal(err)
	}
	wantCommand, err := protocol.CombatPeerPosition(0, clientCommand)
	if err != nil {
		t.Fatal(err)
	}
	want := protocol.CombatData(wantCommand)
	if betaSent[0].ID != want.ID || !bytes.Equal(betaSent[0].Body, want.Body) {
		t.Fatal("relay did not authenticate the sender's combat entity")
	}
	shotFrame, err := hex.DecodeString("0000004f1b21274b4c62634b5835422234234b4b4b4b4b4b4b4d4f4b4b4b4b4a734b4b4b4b4b4b212124462b21212121224b4b4b4b4b4b4b4c62634b5835422234234b582d554b4c606c4b4c61752224714c1b")
	if err != nil {
		t.Fatal(err)
	}
	hub.relayCombatCommands(alpha, shotFrame)
	if len(alphaSent) != 0 || len(otherSent) != 0 || len(betaSent) != 3 {
		t.Fatal("combined position and fire did not stay within the active room")
	}
	shotCommands, err := protocol.ParseClientCombatData(shotFrame)
	if err != nil {
		t.Fatal(err)
	}
	wantFireCommand, err := protocol.CombatPeerFire(0, shotCommands[37:])
	if err != nil {
		t.Fatal(err)
	}
	wantFire := protocol.CombatData(wantFireCommand)
	if betaSent[2].ID != wantFire.ID || !bytes.Equal(betaSent[2].Body, wantFire.Body) {
		t.Fatal("relay did not map native fire with the authenticated entity")
	}
	bad := append([]byte(nil), frame...)
	bad[10] ^= 1
	hub.relayCombatCommands(alpha, bad)
	if len(betaSent) != 3 {
		t.Fatal("relayed corrupt native position")
	}
	hub.sessions[alpha] = planetPresence{callSign: "Alpha", location: state.Location{RoomID: 2}}
	hub.relayCombatCommands(alpha, frame)
	if len(betaSent) != 3 || len(otherSent) != 0 {
		t.Fatal("relayed a pilot after it left the launched room")
	}
}

func TestMissionMembershipRefreshesStagingCountsAndReleasesDisconnectedSeat(t *testing.T) {
	world, err := state.Open(filepath.Join(t.TempDir(), "world.json"))
	if err != nil {
		t.Fatal(err)
	}
	for _, name := range []string{"Stager", "Seated"} {
		if _, err := world.CreatePilot(name); err != nil {
			t.Fatal(err)
		}
		if err := world.JoinPlanet(name, state.SolarisPlanetID); err != nil {
			t.Fatal(err)
		}
	}
	hub := newSessionHub(world)
	hub.planetRoomType = 1
	var updates []protocol.PDU
	stager := &liveSession{sendGamePDU: func(p protocol.PDU) error { updates = append(updates, p); return nil }}
	seated := &liveSession{sendGamePDU: func(protocol.PDU) error { return nil }}
	pilot, _ := world.Pilot("Stager")
	hub.publishPlanetJoin(stager, pilot)
	hub.publishRoomLists()
	rooms, _ := solarisMissionRooms(world, pilot)
	roomID := int(rooms[0].ID.Number)
	peer, _ := world.Pilot("Seated")
	hub.publishPlanetJoin(seated, peer)
	updates = nil
	if err := world.JoinAvailableRoom("Seated", roomID); err != nil {
		t.Fatal(err)
	}
	peer, _ = world.Pilot("Seated")
	rooms, _ = solarisMissionRooms(world, pilot)
	hub.publishMissionJoin(seated, peer, rooms[0], missionCombat(world, rooms[0]), missionSeatUpdates(world, roomID))
	assertRoom := func(wantOccupancy uint32, wantSeats []protocol.SeatLite) {
		t.Helper()
		var got *protocol.PDU
		for i := range updates {
			if updates[i].ID == 0x98 && binary.BigEndian.Uint32(updates[i].Body) == uint32(roomID) {
				got = &updates[i]
			}
		}
		if got == nil {
			t.Fatal("staging client did not receive READY_ROOM_UPDATE")
		}
		infos, _ := solarisMissionRooms(world, pilot)
		if infos[0].Occupancy != wantOccupancy || infos[0].OpenDefenderSeats != 4-byte(wantOccupancy) || infos[0].OpenAttackerSeats != 4 {
			t.Fatalf("staging counts: %+v", infos[0])
		}
		want := protocol.ReadyRoomUpdate(infos[0], wantSeats)
		if !bytes.Equal(got.Body, want.Body) {
			t.Fatal("room update has stale occupants or counts")
		}
	}
	assertRoom(1, []protocol.SeatLite{{Number: 1, Team: 1, CallSign: "Seated"}})
	updates = nil
	hub.remove(seated)
	assertRoom(0, nil)
	peer, _ = world.Pilot("Seated")
	if peer.Location.RoomID != 0 {
		t.Fatalf("disconnected pilot retained room: %+v", peer.Location)
	}
	for _, lance := range world.Snapshot().Rooms[roomID].Seats {
		for _, seat := range lance {
			if seat.CallSign != "" {
				t.Fatalf("ghost seat: %+v", seat)
			}
		}
	}
}

func TestReturningToStagingReannouncesCachedPeers(t *testing.T) {
	hub := newSessionHub()
	hub.planetRoomType = 1
	var received []protocol.PDU
	returning := &liveSession{sendGamePDU: func(p protocol.PDU) error { received = append(received, p); return nil }}
	peer := &liveSession{sendGamePDU: func(protocol.PDU) error { return nil }}
	location := state.Location{PlanetID: 121, CityID: 1}
	hub.publishPlanetJoin(returning, state.Pilot{CallSign: "Returning", Location: location})
	hub.publishPlanetJoin(peer, state.Pilot{CallSign: "AlreadyCached", Location: location})
	previous := planetPresence{callSign: "Returning", location: state.Location{PlanetID: 121, CityID: 1, RoomID: 1}}
	hub.sessions[returning] = previous
	received = nil
	hub.publishMissionDeparture(returning, previous, state.Pilot{CallSign: "Returning", Location: location})
	if len(received) == 0 || received[0].ID != protocol.RoomUpdateID {
		t.Fatal("returning client did not rebuild its staging room")
	}
	arrived := false
	snapshot := false
	for _, p := range received {
		if p.ID == 0x3b {
			snapshot = true
		}
		if p.ID == protocol.OccupantArrivedID && bytes.Equal(p.Body, protocol.OccupantArrivedFor("AlreadyCached").Body) {
			arrived = true
		}
	}
	if !arrived || !snapshot {
		t.Fatal("returning room requires a full occupant snapshot and cached-peer membership")
	}
}

func TestLeavingMissionClearsNativeSeatBeforeLobbyTransition(t *testing.T) {
	world, err := state.Open(filepath.Join(t.TempDir(), "world.json"))
	if err != nil {
		t.Fatal(err)
	}
	if _, err := world.CreatePilot("Leaving"); err != nil {
		t.Fatal(err)
	}
	if err := world.JoinPlanet("Leaving", 121); err != nil {
		t.Fatal(err)
	}
	rooms, _ := world.RoomsForSolarisCity("Leaving")
	if err := world.JoinAvailableRoom("Leaving", rooms[0].ID); err != nil {
		t.Fatal(err)
	}
	pilot, _ := world.Pilot("Leaving")
	previous := planetPresence{callSign: pilot.CallSign, location: pilot.Location}
	hub := newSessionHub(world)
	hub.planetRoomType = 1
	var updates []protocol.PDU
	session := &liveSession{sendGamePDU: func(p protocol.PDU) error { updates = append(updates, p); return nil }}
	hub.sessions[session] = previous
	if err := world.LeaveRoom("Leaving"); err != nil {
		t.Fatal(err)
	}
	pilot, _ = world.Pilot("Leaving")
	hub.publishMissionDeparture(session, previous, pilot)
	if len(updates) < 2 || updates[0].ID != protocol.CombatConfigurationUpdateID || updates[1].ID != protocol.RoomUpdateID {
		t.Fatal("lobby transition omitted the native combat-seat removal")
	}
	if bytes.Contains(updates[0].Body, []byte("Leaving")) {
		t.Fatal("leaving client still sees its own occupied combat seat")
	}
	if !bytes.Equal(updates[0].Body, missionCombat(world, protocol.RoomInfo{ID: protocol.RoomID{Number: uint32(rooms[0].ID), Host: "local"}}).Body) {
		t.Fatal("departing client did not get the updated previous mission")
	}
}

func TestSolarisStagingRosterAndCountStayWithinCity(t *testing.T) {
	hub := newSessionHub()
	hub.planetRoomType = 1
	var firstMessages, secondMessages []protocol.PDU
	first := &liveSession{sendGamePDU: func(p protocol.PDU) error { firstMessages = append(firstMessages, p); return nil }}
	second := &liveSession{sendGamePDU: func(p protocol.PDU) error { secondMessages = append(secondMessages, p); return nil }}
	hub.publishPlanetJoin(first, state.Pilot{CallSign: "CityOne", Location: state.Location{PlanetID: 121, CityID: 1}})
	firstMessages = nil
	hub.publishPlanetJoin(second, state.Pilot{CallSign: "CityTwo", Location: state.Location{PlanetID: 121, CityID: 2}})
	if len(firstMessages) != 0 {
		t.Fatalf("another city's arrival changed the first city's roster: %#v", firstMessages)
	}
	room := protocol.RoomInfo{
		ID: solarisLobbyRoomID(2), Name: "Solaris", Stage: solarisLobbyRoomID(2),
		Occupancy: 1, Capacity: 100, Type: 1, TotalOccupancy: 1,
		AllowPlayerRooms: true, MaxRank: 255, PlanetID: 121, Visible: true,
	}
	if len(secondMessages) == 0 || !bytes.Equal(secondMessages[0].Body, protocol.RoomUpdate(1, true, room).Body) {
		t.Fatal("city lobby count includes occupants from another city")
	}
	var snapshot *protocol.PDU
	for i := range secondMessages {
		if secondMessages[i].ID == 0x3b {
			snapshot = &secondMessages[i]
		}
		if bytes.Contains(secondMessages[i].Body, []byte("CityOne")) {
			t.Fatal("another city's pilot leaked into the new lobby roster")
		}
	}
	if snapshot == nil || !bytes.Equal(snapshot.Body, protocol.UpdateOccupants(room, []string{"CityTwo"}).Body) {
		t.Fatal("city lobby snapshot does not match its membership and count")
	}
	hub.remove(second)
	if len(firstMessages) != 0 {
		t.Fatal("another city's disconnect changed the first city's roster")
	}
}

func TestCombatFireRejectsUnknownRoomTargetsAndWeaponSlots(t *testing.T) {
	// Captured native shot contains no target and uses the first weapon slot.
	body, err := hex.DecodeString("0000004f1b21274b4c62634b5835422234234b4b4b4b4b4b4b4d4f4b4b4b4b4a734b4b4b4b4b4b212124462b21212121224b4b4b4b4b4b4b4c62634b5835422234234b582d554b4c606c4b4c61752224714c1b")
	if err != nil {
		t.Fatal(err)
	}
	commands, err := protocol.ParseClientCombatData(body)
	if err != nil {
		t.Fatal(err)
	}
	roster := map[string]byte{"Alpha": 0, "Beta": 1}
	for _, tc := range []struct {
		name         string
		slot, target byte
		allowed      bool
	}{
		{"untargeted laser", 0, 0, true},
		{"rostered target", 0, 2, true},
		{"last stock weapon", 2, 2, true},
		{"unknown target", 0, 3, false},
		{"absent weapon", 3, 2, false},
		{"large weapon", 84, 0, false},
	} {
		t.Run(tc.name, func(t *testing.T) {
			fire := append([]byte(nil), commands[37:]...)
			fire[1], fire[2] = '!'+tc.slot, '!'+tc.target
			got, err := combatFireForRoom(0, roster, fire)
			if !tc.allowed {
				if err == nil || got.ID != 0 {
					t.Fatal("invalid fire references were relayed")
				}
				return
			}
			if err != nil {
				t.Fatal(err)
			}
			peer, err := protocol.CombatPeerFire(0, fire)
			if err != nil || !bytes.Equal(got.Body, protocol.CombatData(peer).Body) {
				t.Fatal("valid shot changed during validation")
			}
		})
	}
}

func TestStandaloneCombatFireRelaysOnlyAfterReadyToRoomPeer(t *testing.T) {
	data, err := os.ReadFile(filepath.Join("..", "..", "data", "test-fixtures", "combat-live-standalone-fire.json"))
	if err != nil {
		t.Fatal(err)
	}
	var fixture struct {
		ClientBodyHex string `json:"client_body_hex"`
	}
	if err := json.Unmarshal(data, &fixture); err != nil {
		t.Fatal(err)
	}

	if err != nil {
		t.Fatal(err)
	}
	hub := newSessionHub()
	hub.experimentalCombatInit = true
	hub.combatEntities = map[int]map[string]byte{1: {"Alpha": 0, "Beta": 1}, 2: {"Other": 0}}
	hub.combatActive = map[int]bool{1: false}
	var self, peer, elsewhere []protocol.PDU
	alpha := &liveSession{sendGamePDU: func(p protocol.PDU) error { self = append(self, p); return nil }}
	beta := &liveSession{sendGamePDU: func(p protocol.PDU) error { peer = append(peer, p); return nil }}

		t.Fatal("fire relayed before native Ready")

		t.Fatal("standalone shot did not reach only its active-room peer")

	if err != nil {
		t.Fatal(err)
	}
	want, err := combatFireForRoom(0, hub.combatEntities[1], command)
	if err != nil || !bytes.Equal(peer[0].Body, want.Body) {
		t.Fatal("standalone shot changed during relay")
	}

func TestNativeCombatSpawnsFaceOriginAndKeepSeatsSeparate(t *testing.T) {
	seen := map[[2]int32]bool{}
	for side := range 2 {
		for slot := range 4 {
			x, y := nativeCombatSpawn(side, slot)
			if side == 0 && y != 10000 || side == 1 && y != -10000 {
				t.Fatalf("side %d slot %d is on wrong side: %d,%d", side, slot, x, y)
			}
			if seen[[2]int32{x, y}] {
				t.Fatalf("duplicate spawn %d,%d", x, y)
			}
			seen[[2]int32{x, y}] = true
			peerX, peerY := nativeCombatSpawn(1-side, slot)
			if x != -peerX || y != -peerY {
				t.Fatal("opposing seats must face along the same line through the origin")
			}
		}
	}
}

func TestCombatFireRejectsInvalidMechCollisionPart(t *testing.T) {
	data, err := os.ReadFile(filepath.Join("..", "..", "data", "test-fixtures", "combat-native-targeted-fire.json"))
	if err != nil {
		t.Fatal(err)
	}
	var fixture struct {
		CommandHex string `json:"command_hex"`
	}
	if err = json.Unmarshal(data, &fixture); err != nil {
		t.Fatal(err)
	}
	command, err := hex.DecodeString(fixture.CommandHex)
	if err != nil {
		t.Fatal(err)
	}
	for _, part := range []int{-1, 0, 10, 11, 84} {
		fire := append([]byte(nil), command...)
		value := part + 1
		fire[3], fire[4], fire[5] = byte(value/(85*85))+'!', byte(value/85%85)+'!', byte(value%85)+'!'
		_, err := combatFireForRoom(0, map[string]byte{"Alpha": 0, "Beta": 1}, fire)
		if (err == nil) != (part >= 0 && part < 11) {
			t.Fatalf("Mech part %d: %v", part, err)
		}
		// Terrain/untargeted reports use another part space; do not apply the
		// Mech armor bound when the collision entity is absent.
		fire[2] = '!'
		if _, err := combatFireForRoom(0, map[string]byte{"Alpha": 0, "Beta": 1}, fire); err != nil {
			t.Fatalf("non-Mech part %d: %v", part, err)
		}
	}
}
